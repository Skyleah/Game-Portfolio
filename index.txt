    class DreamSoundSynth {
      constructor() {
        this.ctx = null;
        this.enabled = true;
        this.droneOsc1 = null;
        this.droneOsc2 = null;
        this.droneGain = null;
      }

      init() {
        if (!this.ctx) {
          const AudioContext = window.AudioContext || window.webkitAudioContext;
          if (AudioContext) {
            this.ctx = new AudioContext();
            this.startPastelDrone();
          }
        } else if (this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
      }

      startPastelDrone() {
        if (!this.ctx || this.droneGain) return;
        try {
          this.droneGain = this.ctx.createGain();
          this.droneGain.gain.setValueAtTime(0.015, this.ctx.currentTime);
          this.droneGain.connect(this.ctx.destination);

          this.droneOsc1 = this.ctx.createOscillator();
          this.droneOsc1.type = 'sine';
          this.droneOsc1.frequency.setValueAtTime(174.61, this.ctx.currentTime);
          this.droneOsc1.connect(this.droneGain);
          this.droneOsc1.start();

          this.droneOsc2 = this.ctx.createOscillator();
          this.droneOsc2.type = 'triangle';
          this.droneOsc2.frequency.setValueAtTime(261.63, this.ctx.currentTime);
          this.droneOsc2.connect(this.droneGain);
          this.droneOsc2.start();
        } catch (e) {
          console.warn("Audio drone err:", e);
        }
      }

      playSlash() {
        if (!this.enabled || !this.ctx) return;
        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(880, now);
        osc.frequency.exponentialRampToValueAtTime(1320, now + 0.08);
        osc.frequency.exponentialRampToValueAtTime(440, now + 0.2);

        gain.gain.setValueAtTime(0.2, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.2);

        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(now);
        osc.stop(now + 0.21);
      }

      playSparkleImpact() {
        if (!this.enabled || !this.ctx) return;
        const now = this.ctx.currentTime;
        [587.33, 739.99, 880, 1174.66].forEach((f, idx) => {
          const t = now + idx * 0.035;
          const osc = this.ctx.createOscillator();
          const gain = this.ctx.createGain();
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(f, t);

          gain.gain.setValueAtTime(0.22, t);
          gain.gain.exponentialRampToValueAtTime(0.001, t + 0.35);

          osc.connect(gain);
          gain.connect(this.ctx.destination);
          osc.start(t);
          osc.stop(t + 0.36);
        });
      }

      playConfettiPop() {
        if (!this.enabled || !this.ctx) return;
        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(750, now);
        osc.frequency.exponentialRampToValueAtTime(1700, now + 0.12);
        gain.gain.setValueAtTime(0.32, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.18);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(now);
        osc.stop(now + 0.19);
      }

      playRevealHarp() {
        if (!this.enabled || !this.ctx) return;
        const notes = [440, 523.25, 659.25, 783.99, 880, 1046.5];
        notes.forEach((freq, idx) => {
          const t = this.ctx.currentTime + idx * 0.07;
          const osc = this.ctx.createOscillator();
          const gain = this.ctx.createGain();
          osc.type = 'sine';
          osc.frequency.setValueAtTime(freq, t);

          gain.gain.setValueAtTime(0.001, t);
          gain.gain.linearRampToValueAtTime(0.18, t + 0.04);
          gain.gain.exponentialRampToValueAtTime(0.001, t + 0.8);

          osc.connect(gain);
          gain.connect(this.ctx.destination);
          osc.start(t);
          osc.stop(t + 0.85);
        });
      }

      playRabbitHop() {
        if (!this.enabled || !this.ctx) return;
        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(320, now);
        osc.frequency.exponentialRampToValueAtTime(180, now + 0.09);
        gain.gain.setValueAtTime(0.035, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.09);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(now);
        osc.stop(now + 0.1);
      }

      toggle() {
        this.enabled = !this.enabled;
        if (this.droneGain) {
          this.droneGain.gain.setValueAtTime(this.enabled ? 0.015 : 0, this.ctx.currentTime);
        }
        return this.enabled;
      }
    }

      touchId = t.identifier;
      const r = joystickZone.getBoundingClientRect();
      stickCenter = { x: r.left + r.width / 2, y: r.top + r.height / 2 };
    }, { passive: false });

    window.addEventListener('touchmove', (e) => {
      for (let i = 0; i < e.changedTouches.length; i++) {
        const t = e.changedTouches[i];
        if (t.identifier === touchId) {
          const dx = t.clientX - stickCenter.x;
          const dy = t.clientY - stickCenter.y;
          const dist = Math.min(45, Math.hypot(dx, dy));
          const angle = Math.atan2(dy, dx);
          const nx = Math.cos(angle) * dist;
          const ny = Math.sin(angle) * dist;
          joystickKnob.style.transform = `translate(${nx}px, ${ny}px)`;
          stickVec.x = nx / 45;
          stickVec.y = ny / 45;
        }
      }
    }, { passive: true });

    const stopStick = (e) => {
      for (let i = 0; i < e.changedTouches.length; i++) {
        if (e.changedTouches[i].identifier === touchId) {
          touchId = null;
          joystickKnob.style.transform = 'translate(0px, 0px)';
          stickVec.x = 0;
          stickVec.y = 0;
        }
      }
    };

    function triggerFairySlash() {
      if (player.isAttacking) return;
      player.isAttacking = true;
      player.attackTime = 0;
      sound.playSlash();

        detectSlashHit();

    // Hit detection for Props AND smackable Critters (Frogs and Bunnies only)
    function detectSlashHit() {
      const hitRadius = 3.6;
      let struck = false;

      // 1. Interactive Chests/Props
      interactiveProps.forEach(prop => {
        const d = Math.hypot(player.group.position.x - prop.root.position.x, player.group.position.z - prop.root.position.z);
        if (d <= hitRadius) {
          struck = true;
          onPropUnveiled(prop);
        }
      });

      // 2. Smack Frogs and Bunnies only -> Confetti Burst
      smackableCreatures.forEach(critter => {
        if (!critter.active) return;
        const d = Math.hypot(player.group.position.x - critter.group.position.x, player.group.position.z - critter.group.position.z);
        if (d <= hitRadius) {
          critter.active = false;
          critter.group.visible = false;
          triggerCritterConfetti(critter.group.position);
          mouse.pitch += 0.015;

          // Respawn after 6 seconds at random meadow coordinates
          setTimeout(() => {
            const rx = (Math.random() - 0.5) * 32;
            const rz = (Math.random() - 0.5) * 32;
            const ry = getTerrainHeight(rx, rz);
            critter.group.position.set(rx, ry, rz);
            if (critter.type === 'frog') {
              critter.basePos.set(rx, ry, rz);
            }
            critter.group.visible = true;
            critter.active = true;
          }, 6000);
        }
      });

      if (struck) {
        sound.playSparkleImpact();
        mouse.pitch += 0.015;
      }
    }

    function onPropUnveiled(prop) {
      if (!prop.isDiscovered) {
        prop.isDiscovered = true;
        updateBadge();

        spawnSparkleBurst(prop.root.position, prop.data.color);

        if (prop.data.type === 'gift_box' || prop.data.type === 'barrel_moss') {
          prop.isBroken = true;
          prop.objectGroup.visible = false;
        }

        prop.revealTime = 0.01;
        sound.playRevealHarp();

        setTimeout(() => {
          openProjectModal(prop.data);
        }, 750);
      } else {
        openProjectModal(prop.data);
      }
    }

    const clock = new THREE.Clock();

    function updateSimulation(delta) {
      if (isStandardViewOpen) return;

      const elapsed = clock.getElapsedTime();

      // 1. Player Locomotion
      let mx = 0;
      let mz = 0;
      if (keys.w) mz -= 1;
      if (keys.s) mz += 1;
      if (keys.a) mx -= 1;
      if (keys.d) mx += 1;

      if (touchId !== null) {
        mx += stickVec.x;
        mz += stickVec.y;
      }

      const moveLen = Math.hypot(mx, mz);
      const isMoving = moveLen > 0.05;

      if (isMoving) {
        const nx = mx / Math.max(1, moveLen);
        const nz = mz / Math.max(1, moveLen);

        const fwd = new THREE.Vector3(-Math.sin(mouse.yaw), 0, -Math.cos(mouse.yaw));
        const rgt = new THREE.Vector3(Math.cos(mouse.yaw), 0, -Math.sin(mouse.yaw));

        const moveDir = new THREE.Vector3().addScaledVector(rgt, nx).addScaledVector(fwd, nz).normalize();
        const curSpeed = player.speed * (keys.shift ? player.sprintMultiplier : 1.0);

        player.group.position.x += moveDir.x * curSpeed * delta;
        player.group.position.z += moveDir.z * curSpeed * delta;

        const targetRot = Math.atan2(moveDir.x, moveDir.z);
        player.rotationAngle = THREE.MathUtils.lerp(player.rotationAngle, targetRot, 14 * delta);
        heroineMesh.rotation.y = player.rotationAngle;

        skirt.position.y = 0.72 + Math.abs(Math.sin(elapsed * 13)) * 0.08;
        ponytail.rotation.x = -Math.PI / 3 + Math.sin(elapsed * 13) * 0.15;

        const headTrail = trailHistory.shift();
        headTrail.x = player.group.position.x + (Math.random() - 0.5) * 0.3;
        headTrail.y = player.group.position.y + 0.6 + (Math.random() - 0.5) * 0.3;
        headTrail.z = player.group.position.z + (Math.random() - 0.5) * 0.3;
        headTrail.life = 1.0;
        trailHistory.push(headTrail);
      } else {
        skirt.position.y = 0.72 + Math.sin(elapsed * 2.5) * 0.02;
        ponytail.rotation.x = -Math.PI / 3 + Math.sin(elapsed * 2.5) * 0.04;
      }

      const tPos = trailGeo.attributes.position.array;
      for (let i = 0; i < trailCount; i++) {
        const item = trailHistory[i];
        if (item.life > 0) {
          item.life -= delta * 1.4;
          item.y += delta * 0.3;
        }
        tPos[i * 3] = item.x;
        tPos[i * 3 + 1] = item.y;
        tPos[i * 3 + 2] = item.z;
      }
      trailGeo.attributes.position.needsUpdate = true;

      const groundY = getTerrainHeight(player.group.position.x, player.group.position.z);
      player.velocity.y -= 25.0 * delta;
      player.group.position.y += player.velocity.y * delta;

      if (player.group.position.y <= groundY) {
        player.group.position.y = groundY;
        player.velocity.y = 0;
        player.isGrounded = true;
      }

      const pDist = Math.hypot(player.group.position.x, player.group.position.z);
      if (pDist > 32) {
        const rad = Math.atan2(player.group.position.z, player.group.position.x);
        player.group.position.x = Math.cos(rad) * 32;
        player.group.position.z = Math.sin(rad) * 32;
      }

      if (player.isAttacking) {
        player.attackTime += delta;
        const progress = player.attackTime / player.attackDuration;

        if (progress < 1.0) {
          const swing = Math.sin(progress * Math.PI) * 2.1;
          swordPivot.rotation.x = swing;
          swordPivot.rotation.y = -swing * 0.7;
          slashArc.material.opacity = Math.sin(progress * Math.PI) * 0.9;
        } else {
          player.isAttacking = false;
          swordPivot.rotation.set(0, 0, 0);
          slashArc.material.opacity = 0;
        }
      }

      camLookTarget.copy(player.group.position);
      camLookTarget.y += 1.5;

      const cDist = mouse.distance;
      const cY = camLookTarget.y + Math.sin(mouse.pitch) * cDist;
      const cHoriz = Math.cos(mouse.pitch) * cDist;
      const cX = camLookTarget.x + Math.sin(mouse.yaw) * cHoriz;
      const cZ = camLookTarget.z + Math.cos(mouse.yaw) * cHoriz;

      const safeCY = Math.max(cY, getTerrainHeight(cX, cZ) + 0.5);
      camera.position.set(cX, safeCY, cZ);
      camera.lookAt(camLookTarget);

      butterflies.forEach(b => {
        b.angle += delta * b.speed * 0.5;
        b.group.position.x = b.center.x + Math.sin(b.angle) * 3.5;
        b.group.position.z = b.center.z + Math.cos(b.angle) * 3.5;
        b.group.position.y = b.center.y + Math.sin(elapsed * 3 + b.wingPhase) * 0.5;

        const flap = Math.sin(elapsed * 16 + b.wingPhase) * 0.9;
        b.leftWing.rotation.y = flap;
        b.rightWing.rotation.y = -flap;
        b.group.rotation.y = -b.angle;
      });

      rabbits.forEach(r => {
        if (!r.active) return;
        r.hopTimer -= delta;
        if (r.hopTimer <= 0 && !r.isHopping) {
          r.isHopping = true;
          r.hopProgress = 0;
          r.startPos.copy(r.group.position);
          r.heading += (Math.random() - 0.5) * 1.5;
          const hopDist = 1.4 + Math.random() * 0.8;
          r.targetPos.set(
            r.startPos.x + Math.sin(r.heading) * hopDist,
            0,
            r.startPos.z + Math.cos(r.heading) * hopDist
          );
          if (Math.hypot(r.targetPos.x, r.targetPos.z) > 28) {
            r.heading += Math.PI;
            r.targetPos.set(r.startPos.x, 0, r.startPos.z);
          }
          r.targetPos.y = getTerrainHeight(r.targetPos.x, r.targetPos.z);
          sound.playRabbitHop();
        }

        if (r.isHopping) {
          r.hopProgress += delta / r.hopDuration;
          if (r.hopProgress >= 1) {
            r.hopProgress = 1;
            r.isHopping = false;
            r.hopTimer = 1.5 + Math.random() * 3.5;
          }
          const t = r.hopProgress;
          r.group.position.lerpVectors(r.startPos, r.targetPos, t);
          r.group.position.y += Math.sin(t * Math.PI) * 0.6;
          r.group.rotation.y = r.heading;
        }
      });

      frogs.forEach(f => {
        if (!f.active) return;
        f.hopCooldown -= delta;
        if (f.hopCooldown <= 0 && !f.isHopping) {
          f.isHopping = true;
          f.hopProgress = 0;
        }
        if (f.isHopping) {
          f.hopProgress += delta * 2.8;
          f.group.position.y = f.basePos.y + Math.sin(f.hopProgress * Math.PI) * 0.5;
          if (f.hopProgress >= 1) {
            f.isHopping = false;
            f.group.position.y = f.basePos.y;
            f.hopCooldown = 3 + Math.random() * 6;
          }
        }
      });

      let nearestDist = Infinity;
      let targetItem = null;

      interactiveProps.forEach(prop => {
        const d = Math.hypot(player.group.position.x - prop.root.position.x, player.group.position.z - prop.root.position.z);
        if (d < nearestDist) {
          nearestDist = d;
          targetItem = prop;
        }

        prop.label.lookAt(camera.position);

        if (prop.revealTime > 0) {
          prop.revealTime = Math.min(1.0, prop.revealTime + delta * 1.6);
          const t = prop.revealTime;
          prop.artifact.visible = true;
          prop.artifact.position.y = 0.5 + Math.sin(t * Math.PI * 0.5) * 2.2;
          const s = Math.sin(t * Math.PI * 0.5);
          prop.artifact.scale.set(s, s, s);
          prop.artifact.rotation.y += delta * 1.8;
          prop.artLight.intensity = t * 2.5;

          if (prop.lid && !prop.isBroken) {
            prop.lid.rotation.x = -Math.sin(t * Math.PI * 0.5) * 1.6;
          }
        } else if (prop.isDiscovered) {
          prop.artifact.rotation.y += delta * 0.8;
          prop.artifact.position.y = 2.7 + Math.sin(elapsed * 2.5) * 0.12;
        }
      });

      if (nearestDist < 3.8 && targetItem && !isModalOpen) {
        promptElem.classList.remove('opacity-0');
        crosshair.classList.add('active');
      } else {
        promptElem.classList.add('opacity-0');
        crosshair.classList.remove('active');
      }

      for (let i = sparkleBurstDebris.length - 1; i >= 0; i--) {
        const sp = sparkleBurstDebris[i];
        sp.vel.y -= 14 * delta;
        sp.mesh.position.addScaledVector(sp.vel, delta);
        sp.mesh.rotation.x += delta * 6;
        sp.life -= delta * 0.7;

        if (sp.life <= 0) {
          scene.remove(sp.mesh);
          sp.mesh.geometry.dispose();
          sparkleBurstDebris.splice(i, 1);
        }
      }

      // Confetti physics updates
      for (let c = confettiParticles.length - 1; c >= 0; c--) {
        const cp = confettiParticles[c];
        cp.vel.y -= 13 * delta;
        cp.mesh.position.addScaledVector(cp.vel, delta);
        cp.mesh.rotation.x += cp.rotVel.x * delta;
        cp.mesh.rotation.y += cp.rotVel.y * delta;
        cp.life -= delta * 0.65;

        if (cp.life <= 0) {
          scene.remove(cp.mesh);
          cp.mesh.geometry.dispose();
          confettiParticles.splice(c, 1);
        }
      }

      const pArr = petalGeo.attributes.position.array;
      for (let p = 0; p < petalCount; p++) {
        pArr[p * 3] += petalVelocities[p].x * delta;
        pArr[p * 3 + 1] += petalVelocities[p].y * delta;
        pArr[p * 3 + 2] += petalVelocities[p].z * delta;

        if (pArr[p * 3 + 1] < 0.2) {
          pArr[p * 3 + 1] = 12 + Math.random() * 4;
          pArr[p * 3] = (Math.random() - 0.5) * 60;
          pArr[p * 3 + 2] = (Math.random() - 0.5) * 60;
        }
      }
      petalGeo.attributes.position.needsUpdate = true;

      centralHeart.rotation.y += delta * 0.5;
      orbitRing.rotation.z += delta * 0.3;
    }

    window.addEventListener('resize', () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    });

    window.onload = function () {
      function renderLoop() {
        requestAnimationFrame(renderLoop);
        const dt = Math.min(clock.getDelta(), 0.08);
        updateSimulation(dt);
        if (!isStandardViewOpen) {
          renderer.render(scene, camera);
        }
      }
      renderLoop();
    };
